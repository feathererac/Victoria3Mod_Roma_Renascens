	name = "je_rome_roman_spark_button"
	desc = "je_rome_roman_spark_button_desc"

	visible = {
		OR = {
			c:GRE ?= this
			c:BYZ ?= this
		}
		has_journal_entry = je_greek_nationalism
		NOT = { has_variable = rome_roman_spark_taken }
	}

	possible = {
	}

	ai_chance = {
		value = 0
	}

	effect = {
		# 罗马火种军事修正（攻防+30/杀伤+30%/士气打击+30%），持续10年
		add_modifier = {
			name = rome_roman_spark
			months = 120
		}
		# 奥斯曼地区所有故土的宣称
		rome_add_ottoman_claims = yes
		# 标记已点燃，按钮与议程消失
		set_variable = { name = rome_roman_spark_taken }
	}
}
