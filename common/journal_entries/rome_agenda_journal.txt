### 重建罗马 - 点击后变为持续日志，授予「重建中的罗马」；全部罗马疆域到手后完成并授予「永恒罗马」 ###
je_rome_declare_imperium = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		OR = {
			c:BYZ ?= this
			c:SPQR ?= this
		}
	}

	possible = {
		OR = {
			c:BYZ ?= this
			c:SPQR ?= this
		}
	}

	# 点击按钮后并不消失，而是成为持续重建日志；只有当全部罗马疆域都掌握在手时才完成
	complete = {
		hidden_trigger = { rome_all_held = yes }
	}

	# 完成（全部疆域到手）：移除重建中的罗马各等级与条条大路通罗马，授予永恒罗马
	on_complete = {
		rome_remove_all_rebuilding_modifiers = yes
		add_modifier = { name = rome_imperium_aeternum }
	}

	# 按钮：宣布重建罗马（一次性，点击后按钮因 rome_roman_redeclared_complete 变量而隐藏，日志保留）
	scripted_button = je_rome_declare_imperium_button

	weight = 200

	should_be_pinned_by_default_uninvolved_or_context = yes
}

### 希腊觉醒 - 科技启蒙与文化复兴 ###
je_rome_greek_awakening = {
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		OR = {
			c:GRE ?= this
			c:BYZ ?= this
			c:SPQR ?= this
		}
		NOT = { has_variable = rome_greek_awakening_completed }
	}

	possible = {
		OR = {
			c:GRE ?= this
			c:BYZ ?= this
			c:SPQR ?= this
		}
	}

	immediate = {
		add_modifier = {
			name = rome_innovation_boost
			months = 120
		}
		add_modifier = {
			name = rome_literacy_boost
		}
		add_modifier = {
			name = rome_greek_treasury_boost
		}
	}

	complete = {
		literacy_rate >= 0.7
	}

	on_complete = {
		if = { limit = { has_modifier = rome_innovation_boost } remove_modifier = rome_innovation_boost }
		if = { limit = { has_modifier = rome_literacy_boost } remove_modifier = rome_literacy_boost }
		add_modifier = {
			name = rome_literacy_tech_boost
		}
		set_variable = { name = rome_greek_awakening_completed }
	}

	scripted_button = je_rome_greek_awakening_loyalists_button

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}
