### 重建罗马 - 点击后变为持续日志，授予「重建中的罗马」；全部罗马疆域到手后完成并授予「永恒罗马」 ###
je_rome_declare_imperium = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		OR = {
			c:BYZ ?= this
			c:SPQR ?= this
		}
	}

	possible = {
		OR = {
			c:BYZ ?= this
			c:SPQR ?= this
		}
	}

	# 点击按钮后立即完成消失（变身罗马功能在 scripted_effect 里实现）
	complete = {
		hidden_trigger = { has_variable = rome_roman_redeclared_complete }
	}

	# 变身罗马后不再需要 on_complete 逻辑，buff 管理已转移到奥古斯都
	on_complete = {
	}

	# 按钮：宣布重建罗马（一次性，点击后按钮因 rome_roman_redeclared_complete 变量而隐藏，日志保留）
	scripted_button = je_rome_declare_imperium_button

	weight = 200

	should_be_pinned_by_default_uninvolved_or_context = yes
}

		OR = {
			c:BYZ ?= this
			c:SPQR ?= this
		}

	}

	possible = {
		OR = {
			c:BYZ ?= this
			c:SPQR ?= this
		}
		NOT = { has_variable = rome_augustus_complete }
	}

	# 点击「加冕奥古斯都」（所有地区宣称并征服）后置位变量 → 议程完成消失

		hidden_trigger = { has_variable = rome_augustus_complete }
	}

	# 动态状态描述：悬停时可高亮对应地理区域

		first_valid = {
			triggered_desc = {
				desc = je_rome_augustus_status_egypt
				trigger = { NOT = { has_variable = rome_augustus_egypt_done } }
			}
			triggered_desc = {
				desc = je_rome_augustus_status_africa
				trigger = { NOT = { has_variable = rome_augustus_africa_done } }
			}
			triggered_desc = {
				desc = je_rome_augustus_status_italy
				trigger = { NOT = { has_variable = rome_augustus_italy_claimed } }
			}
			triggered_desc = {
				desc = je_rome_augustus_status_conquer_italy
				trigger = {
					has_variable = rome_augustus_italy_claimed
					NOT = { has_variable = rome_augustus_italy_done }
				}
			}
			triggered_desc = {
				desc = je_rome_augustus_status_conquer_hispania
				trigger = {
					has_variable = rome_augustus_hispania_claimed
					NOT = { has_variable = rome_augustus_hispania_done }

			triggered_desc = {

				trigger = {
					has_variable = rome_augustus_gallia_claimed
					NOT = { has_variable = rome_augustus_gallia_done }

			triggered_desc = {
				desc = je_rome_augustus_status_conquer_britannia
				trigger = {
					has_variable = rome_augustus_britannia_claimed
					NOT = { has_variable = rome_augustus_britannia_done }

				desc = je_rome_augustus_status_final
				trigger = {

	scripted_button = je_rome_augustus_egypt_button
	scripted_button = je_rome_augustus_africa_button

	}

		}
		NOT = { has_variable = rome_senate_complete }
	}

	# 三项改革（混合共和制 / 三级权力分配 / 万民法）全部立法完成 → 议程完成

	scripted_button = je_rome_senate_button
	scripted_button = je_rome_senate_three_tier_button
	scripted_button = je_rome_senate_jus_gentium_button

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}

### 希腊觉醒 - 科技启蒙与文化复兴 ###
je_rome_greek_awakening = {
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		OR = {
			c:GRE ?= this
			c:BYZ ?= this
			c:SPQR ?= this
		}
		NOT = { has_variable = rome_greek_awakening_completed }
	}

	possible = {
		OR = {
			c:GRE ?= this
			c:BYZ ?= this
			c:SPQR ?= this
		}
	}

	immediate = {
		add_modifier = {
			name = rome_innovation_boost
			months = 120
		}
		add_modifier = {
			name = rome_literacy_boost
		}
		add_modifier = {
			name = rome_greek_treasury_boost
		}
	}

	complete = {
		literacy_rate >= 0.7
	}

	on_complete = {
		if = { limit = { has_modifier = rome_innovation_boost } remove_modifier = rome_innovation_boost }
		if = { limit = { has_modifier = rome_literacy_boost } remove_modifier = rome_literacy_boost }
		add_modifier = {
			name = rome_literacy_tech_boost
		}
		set_variable = { name = rome_greek_awakening_completed }
	}

	scripted_button = je_rome_greek_awakening_loyalists_button

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}
