### 罗马三级权力分配 - 元老院议程解锁的选举制度 ###
### 三级：贵族/教士/军官（第一级）、工程师/学者（第二级）、其他公民（第三级） ###
law_byzantine_three_tier = {
	group = lawgroup_distribution_of_power

	icon = "gfx/interface/icons/law_icons/wealth_voting.dds"

	progressiveness = -10

	disallowing_laws = {
		law_chiefdom
		law_command_economy
	}

	is_visible = {
		# 仅通过元老院议程解锁，不在普通法律界面显示
		always = no
	}

	on_enact = {
		# 初始化选举信心（选举制度必须）
		if = {
			limit = {
				game_date > 1836.1.1
				country_has_voting_franchise = yes
			}
			initialize_electoral_confidence = yes
		}
		scripted_effect_parties_emergence = yes
		inherit_caciquismo_effect = yes
		inherit_free_elections_effect = yes
	}

	modifier = {
		# +100权威力
		country_authority_add = 100
		# 贵族、教士、军官政治力量+100%
		country_aristocrats_pol_str_mult = 1.0
		country_clergymen_pol_str_mult = 1.0
		country_officers_pol_str_mult = 1.0
		# 工程师、学者政治力量+33%
		country_engineers_pol_str_mult = 0.33
		country_academics_pol_str_mult = 0.33
		# +60基于政府势力的合法性
		country_legitimacy_govt_total_clout_add = 60
		# +10投票的财富门槛
		country_voting_power_wealth_threshold_add = 10
		# +85基于选票的合法性
		country_legitimacy_govt_total_votes_add = 85
		# +1政府规模限额
		country_legitimacy_govt_size_add = 1

		# 利益集团的法律支持由罗马化理念（ideology）提供，
		# 见 common/ideologies/rome_ig_ideologies.txt
	}

	ai_will_do = {
		always = no
	}
}
